variable = value
label:
[label:] command [arguments]
[label:] command

TOKENS
command ::= opcode | pseudo-opcode | macro
opcode ::= LDA | STA | RTS etc.
pseudo-opcode ::= .BYTE | .WORD | .INCLUDE
name ::= letter [letter | digit | dot]{0..15}
label ::= (name) colon
macro ::= ampersand (name)
letter ::= [A-Z]
digit ::= [0-9]
dot ::= '.'
equals ::= '='
star ::= '*'
colon ::= ':'
ampersand ::= '&'

SYNTAX
line ::= assignment | set-location | bare-label | instruction
assignment ::= label equals number
set-location ::= star equals number
bare-label ::= label-prefix
label-prefix ::= label colon
instruction ::= [label-prefix] command [argument]
arguments ::= immediate-arg | direct-arg | indirect-arg | index-arg | indirect-index-arg
